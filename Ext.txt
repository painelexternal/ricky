local COLOR_PROPS = {
	BackgroundColor3 = true, TextColor3 = true, Color = true, ImageColor3 = true,
	ScrollBarImageColor3 = true, PlaceholderColor3 = true,
}
local Registry = setmetatable({}, { __mode = "k" })
local Grads = setmetatable({}, { __mode = "k" })

local function kp(t, c)
	return ColorSequenceKeypoint.new(t, c)
end

local function gradSeq(kind)
	if kind == "text" then
		return ColorSequence.new({ kp(0, C.Text), kp(0.5, C.Text:Lerp(C.Dim, 0.55)), kp(1, C.Text) })
	elseif kind == "shine" then
		return ColorSequence.new({ kp(0, C.Dim), kp(0.5, C.Text), kp(1, C.Dim) })
	elseif kind == "bright" then
		local m = C.Lo:Lerp(C.Hi, 0.62)
		return ColorSequence.new({ kp(0, m), kp(0.5, C.Hi), kp(1, m) })
	elseif kind == "bg" then
		return ColorSequence.new({ kp(0, C.Panel2), kp(1, C.Panel) })
	elseif kind == "main" then
		return ColorSequence.new({ kp(0, C.Panel2), kp(0.5, C.Bg), kp(1, C.Bg) })
	end
	return ColorSequence.new({
		kp(0, C.Lo), kp(0.28, C.Hi), kp(0.5, C.Lo:Lerp(C.Hi, 0.35)), kp(0.76, C.Hi), kp(1, C.Lo),
	})
end

local function metalGradient(parent, kind, rotation)
	local g = Instance.new("UIGradient")
	g.Rotation = rotation or 0
	Grads[g] = kind or "metal"
	g.Color = gradSeq(kind or "metal")
	g.Parent = parent
	return g
end

local function applyThemeColors()
	for inst, list in pairs(Registry) do
		for _, e in ipairs(list) do
			pcall(function()
				inst[e[1]] = C[e[2]]
			end)
		end
	end
	for g, kind in pairs(Grads) do
		g.Color = gradSeq(kind)
	end
end

local function new(class, props, kids)
	props = props or {}
	local inst = Instance.new(class)
	for k, v in pairs(props) do
		if k ~= "Parent" and k ~= "Face" then
			if COLOR_PROPS[k] and type(v) == "string" then
				local role = string.sub(v, 2)
				inst[k] = C[role]
				local list = Registry[inst]
				if not list then
					list = {}
					Registry[inst] = list
				end
				table.insert(list, { k, role })
			else
				inst[k] = v
			end
		end
	end
	if TEXT_CLASSES[class] then
		pcall(function()
			inst.FontFace = FONT[props.Face or "Semi"]
		end)
	end
	for _, child in ipairs(kids or {}) do
		child.Parent = inst
	end
	if props.Parent then
		inst.Parent = props.Parent
	end
	return inst
end

local function isInsideInteractive(guiObj, pos)
	if not guiObj or not guiObj.Visible then
		return false
	end
	if guiObj:IsA("TextButton") or guiObj:IsA("ImageButton") or guiObj:IsA("TextBox") or guiObj:IsA("ScrollingFrame") then
		local ap, as = guiObj.AbsolutePosition, guiObj.AbsoluteSize
		if pos.X >= ap.X and pos.X <= ap.X + as.X and pos.Y >= ap.Y and pos.Y <= ap.Y + as.Y then
			return true
		end
	end
	for _, child in ipairs(guiObj:GetChildren()) do
		if child:IsA("GuiObject") and isInsideInteractive(child, pos) then
			return true
		end
	end
	return false
end

local function toHex(c, a)
	local s = string.format("#%02X%02X%02X", round(c.R * 255), round(c.G * 255), round(c.B * 255))
	if a then
		s = s .. string.format("%02X", round(a * 255))
	end
	return s
end

local function fromHex(txt)
	local clean = (string.gsub(tostring(txt), "#", ""))
	if (#clean ~= 6 and #clean ~= 8) or not string.match(clean, "^%x+$") then
		return nil
	end
	local c = Color3.fromRGB(
		tonumber(string.sub(clean, 1, 2), 16),
		tonumber(string.sub(clean, 3, 4), 16),
		tonumber(string.sub(clean, 5, 6), 16)
	)
	local a = (#clean == 8) and (tonumber(string.sub(clean, 7, 8), 16) / 255) or 1
	return c, a
end

local function makeValue(window, flag, default, apply)
	local obj = { Value = default, _cbs = {}, Flag = flag }
	function obj:OnChanged(fn)
		table.insert(self._cbs, fn)
		return self
	end
	function obj:Set(v, silent)
		self.Value = v
		window.Flags[flag] = v
		apply(v)
		if not silent then
			for _, fn in ipairs(self._cbs) do
				task.spawn(fn, v)
			end
		end
	end
	window.Elements[flag] = obj
	window.Flags[flag] = default
	return obj
end

local function resolveKey(name)
	if name == nil or name == "" then
		return nil
	end
	if typeof(name) == "EnumItem" then
		return name
	end
	local ok, k = pcall(function()
		return Enum.KeyCode[name]
	end)
	if ok and k then
		return k
	end
	ok, k = pcall(function()
		return Enum.UserInputType[name]
	end)
	if ok and k then
		return k
	end
	return nil
end

local function normalizeAssetId(value)
	if value == nil then
		return nil
	end
	if type(value) == "number" then
		return "rbxassetid://" .. tostring(math.floor(value))
	end
	local s = string.gsub(tostring(value), "%s+", "")
	if s == "" then
		return nil
	end
	if string.match(s, "^%d+$") then
		return "rbxassetid://" .. s
	end
	local id = string.match(s, "^rbxassetid://(%d+)$") or string.match(s, "[?&]id=(%d+)")
	if id then
		return "rbxassetid://" .. id
	end
	if string.match(s, "^rbxasset://") or string.match(s, "^rbxthumb://") or string.match(s, "^rbxgameasset://") then
		return s
	end
	return nil
end

local ICON_ORDER = {
	"aim", "eye", "sun", "pin", "gear", "folder", "user", "list", "bolt", "shield", "star", "diamond",
}
local ICON_SET = {}
for _, n in ipairs(ICON_ORDER) do
	ICON_SET[n] = true
end

local function parseIconSpec(spec)
	spec = tostring(spec or "folder")
	local rest = string.match(spec, "^[iI][dD]:(.+)$") or string.match(spec, "^[iI][mM][gG]:(.+)$")
	if not rest and (string.match(spec, "^%d%d%d+$") or string.find(spec, "rbxassetid", 1, true)) then
		rest = spec
	end
	if rest then
		local img = normalizeAssetId(rest)
		if img then
			return "image", img
		end
	end
	return "kind", string.lower(spec)
end

local function makeIcon(parent, spec)
	local mode, value = parseIconSpec(spec)
	if mode == "kind" and not ICON_SET[value] then
		return nil
	end
	local parts = {}
	local holder = new("Frame", {
		AnchorPoint = Vector2.new(0, 0.5), Position = UDim2.new(0, -26, 0.5, 0), Size = UDim2.fromOffset(18, 18),
		BackgroundTransparency = 1, Name = "IconHolder", Parent = parent,
	})
	local mid = Vector2.new(0.5, 0.5)
	local CIRCLE = UDim.new(0.5, 0)
	local R1, R2 = UDim.new(0, 1), UDim.new(0, 2)

	local function solid(x, y, w, h, rot, radius)
		local f = new("Frame", {
			AnchorPoint = mid, Position = UDim2.fromOffset(x, y), Size = UDim2.fromOffset(w, h),
			Rotation = rot or 0, BackgroundColor3 = C.Dim, BorderSizePixel = 0, Parent = holder,
		})
		if radius then
			new("UICorner", { CornerRadius = radius, Parent = f })
		end
		table.insert(parts, { f, "BackgroundColor3" })
	end
	local function outline(x, y, w, h, thick, radius, rot)

			AnchorPoint = mid, Position = UDim2.fromOffset(x, y), Size = UDim2.fromOffset(w, h),
			Rotation = rot or 0, BackgroundTransparency = 1, BorderSizePixel = 0, Parent = holder,
		})

		end
		local s = new("UIStroke", {
			Color = C.Dim, Thickness = thick, ApplyStrokeMode = Enum.ApplyStrokeMode.Border, Parent = f,
		})
		table.insert(parts, { s, "Color" })
	end

	if mode == "image" then
		local img = new("ImageLabel", {
			Size = UDim2.fromScale(1, 1), BackgroundTransparency = 1, Image = value,
			ImageColor3 = C.Dim, ScaleType = Enum.ScaleType.Fit, Parent = holder,
		}, { corner(3) })
		table.insert(parts, { img, "ImageColor3" })
	elseif value == "aim" then
		outline(9, 9, 12, 12, 1.5, CIRCLE)
		solid(9, 9, 4, 4, 0, CIRCLE)
		solid(9, 2, 2, 5)
		solid(9, 16, 2, 5)
		solid(2, 9, 5, 2)
		solid(16, 9, 5, 2)
	elseif value == "eye" then
		outline(9, 9, 18, 11, 1.8, CIRCLE)
		outline(9, 9, 8, 8, 1.8, CIRCLE)
		solid(9, 9, 3.4, 3.4, 0, CIRCLE)
	elseif value == "sun" then
		outline(9, 9, 8, 8, 1.8, CIRCLE)
		for i = 0, 7 do
			local a = math.rad(i * 45)
			solid(9 + math.sin(a) * 7.4, 9 - math.cos(a) * 7.4, 1.8, 3.2, i * 45)
		end
	elseif value == "pin" then
		solid(9, 14.5, 6, 6, 45, R1)
		outline(9, 7.5, 10, 10, 2, CIRCLE)
		solid(9, 7.5, 3, 3, 0, CIRCLE)
	elseif value == "gear" then
		outline(9, 9, 9, 9, 2, CIRCLE)
		for i = 0, 7 do
			local a = math.rad(i * 45)
			solid(9 + math.sin(a) * 6.8, 9 - math.cos(a) * 6.8, 2.4, 4, i * 45)
		end
	elseif value == "folder" then
		solid(5, 4.5, 8, 5, 0, R1)
		outline(9, 11, 16, 11, 1.5, R2)
		solid(9, 11, 8, 2)
	elseif value == "user" then
		solid(9, 5.5, 7, 7, 0, CIRCLE)
		solid(9, 14, 14, 7, 0, CIRCLE)
	elseif value == "list" then
		solid(9, 5, 14, 2.5, 0, R1)
		solid(9, 9, 14, 2.5, 0, R1)
		solid(9, 13, 14, 2.5, 0, R1)

function Library:ApplyTheme(name)
	local theme = THEMES[name]
	if not theme then
		return false
	end
	for role, color in pairs(theme) do
		C[role] = color
	end
	applyThemeColors()
	for _, fn in ipairs(self.Restylers) do
		pcall(fn)
	end
	return true
end

function Library:_tip(obj, str)
	if not str or str == "" then
		return
	end
	local token = 0
	self:_connect(obj.MouseEnter, function()
		token += 1
		local mine = token
		task.delay(0.4, function()
			if mine == token and not self.Destroyed then
				self.TipLabel.Text = str
				self.Tip.Visible = true
			end
		end)
	end)
	self:_connect(obj.MouseLeave, function()
		token += 1
		self.Tip.Visible = false
	end)
end

	local dragging, dragStart, startPos = false, nil, nil
	local function beginDrag(input, allowBody)
		if not isPress(input) or not main.Visible then
			return
		end
		if allowBody and (isInsideInteractive(self.Content, input.Position) or isInsideInteractive(self.Rail, input.Position)) then
			return
		end
		dragging, dragStart, startPos = true, input.Position, main.Position
		local c
		c = input.Changed:Connect(function()
			if input.UserInputState == Enum.UserInputState.End then
				dragging = false
				c:Disconnect()
			end
		end)
	end

		beginDrag(input, false)
	end)

			resizing, rStart, rSize = true, input.Position, self.Size
		end
	end)
	self:_connect(UserInputService.InputChanged, function(input)
		if dragging and dragStart and isMove(input) then
			local d = (input.Position - dragStart) / math.max(uiScale.Scale, 0.01)
			main.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + d.X, startPos.Y.Scale, startPos.Y.Offset + d.Y)
		elseif resizing and isMove(input) then
			local d = (input.Position - rStart) / math.max(uiScale.Scale, 0.01)
			local nx = math.clamp(rSize.X + d.X, MIN_W, MAX_W)
			local ny = math.clamp(rSize.Y + d.Y, MIN_H, MAX_H)
			self.Size = Vector2.new(nx, ny)
			main.Size = UDim2.fromOffset(nx, ny)
		end
	end)
	self:_connect(UserInputService.InputEnded, function(input)
		if isPress(input) then
			dragging, resizing = false, false
		end
	end)

	self:_connect(UserInputService.InputBegan, function(input, gp)
		if input.KeyCode == Enum.KeyCode.Escape and self.Palette.Visible then
			self:TogglePalette(false)
			return
		end
		if gp then
			return
		end
		if input.KeyCode == self.ToggleKey then
			self:SetVisible(not self.Visible)
		elseif input.KeyCode == Enum.KeyCode.K
			and (UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) or UserInputService:IsKeyDown(Enum.KeyCode.RightControl)) then
			self:TogglePalette(not self.Palette.Visible)
		end
	end)

	---------------- animacao de onda metalica + fps
	local clock, acc, frames, barClock, hudClock = 0, 0, 0, 0, 0
	self:_connect(RunService.RenderStepped, function(dt)
		acc += dt
		frames += 1
		if acc >= 0.5 then
			self.FPS = round(frames / acc)
			self.Ping = getPing()
			self.FPSLabel.Text = string.format("%d FPS | %dms", self.FPS, self.Ping)
			acc, frames = 0, 0
		end
		if self.HUDOn then
			hudClock += dt
			if hudClock >= 0.25 then
				hudClock = 0
				self.HUDTop.Text = string.format("RICKY V8.2  |  %d FPS  |  %dms", self.FPS, self.Ping)
				local n = 0
				for _, el in pairs(self.Elements) do
					if el.IsToggle and el.Listed and el.Value == true then
						n += 1
						if n <= 10 then
							self.HUDLines[n].Text = string.upper(el.Label or el.Flag)
							self.HUDLines[n].Visible = true
						end
					end
				end
				for i = n + 1, 10 do
					self.HUDLines[i].Visible = false
				end
			end
		end
		if not self.Visible or not self.WaveOn then
			return
		end
		clock += dt * self.WaveSpeed
		for _, w in ipairs(self.Waves) do
			w[1].Offset = Vector2.new(math.sin(clock * w[2] + w[3]) * w[4], 0)
		end
		for _, g in ipairs(self.Spins) do
			g.Rotation = (clock * 28) % 360
		end
		if self.SheenOn then
			sheenG.Offset = Vector2.new(math.sin(clock * 0.45), 0)
			sheen.Visible = true
		else
			sheen.Visible = false
		end
		barClock += dt
		if barClock >= 1 / 30 then
			barClock = 0
			for i, bar in ipairs(self.Bars) do
				local a = math.sin(clock * 3.2 + i * 0.42) * 0.5 + 0.5
				local b = math.sin(clock * 1.7 - i * 0.21) * 0.5 + 0.5
				bar.Size = UDim2.fromOffset(2, 3 + (a * 0.65 + b * 0.35) * 19)
			end
		end
	end)

function Library:TogglePalette(state)
	local pal = self.Palette
	if state then
		pal.Visible = true
		self.PaletteBox.Text = ""
		self.PaletteFill()
		task.defer(function()
			if not self.Destroyed then
				self.PaletteBox:CaptureFocus()
			end
		end)
	else
		self.PaletteBox:ReleaseFocus()
		tween(pal, 0.15, { Size = UDim2.fromOffset(390, 0) })
		task.delay(0.16, function()
			if not self.Destroyed and pal.Size.Y.Offset <= 1 then
				pal.Visible = false
			end
		end)
	end
end

function Library:Goto(entry)
	self:SelectTab(entry.tab)
	task.delay(0.05, function()
		local page = entry.tab.Page
		if self.Destroyed or not page or not entry.row.Parent then
			return
		end
		local y = entry.row.AbsolutePosition.Y - page.AbsolutePosition.Y + page.CanvasPosition.Y
		page.CanvasPosition = Vector2.new(0, math.max(0, y - 8))
		local st = entry.row:FindFirstChildOfClass("UIStroke")
		if st then
			tween(st, 0.2, { Color = C.Hi })
			task.delay(0.8, function()
				if st.Parent then
					tween(st, 0.4, { Color = C.Line })
				end
			end)
		end
	end)
end

function Library:SetHUD(state)
	self.HUDOn = state
	self.HUD.Visible = state
end

function Library:SetToggleKey(key)
	local k = resolveKey(key)
	if k and k.EnumType == Enum.KeyCode then
		self.ToggleKey = k
	end
end

function Library:SetVisible(state)
	self.Visible = state
	if state then
		self.Main.Visible = true
		self.Launcher.Visible = false
		self.UIScale.Scale = self.BaseScale * 0.9
		tween(self.UIScale, 0.35, { Scale = self.BaseScale }, Enum.EasingStyle.Back)
	else
		tween(self.UIScale, 0.2, { Scale = self.BaseScale * 0.9 })
		task.delay(0.2, function()
			if not self.Visible and not self.Destroyed then
				self.Main.Visible = false
				self.Launcher.Visible = true
			end
		end)
	end
end

function Library:SetMinimized(state)
	self.Minimized = state
	self.Grip.Visible = not state
	if state then
		self.Body.Visible = false
		tween(self.Main, 0.3, { Size = UDim2.fromOffset(self.Size.X, HEADER) })
	else
		self.Body.Visible = true
		tween(self.Main, 0.3, { Size = UDim2.fromOffset(self.Size.X, self.Size.Y) })
	end
end

function Library:SetScale(v)
	self.BaseScale = v
	tween(self.UIScale, 0.2, { Scale = v })
end

function Library:Center()
	tween(self.Main, 0.3, { Position = UDim2.fromScale(0.5, 0.5) })
end

function Library:Destroy()
	if self.Destroyed then
		return
	end
	self.Destroyed = true
	for _, fn in ipairs(self.Cleanups) do
		pcall(fn)
	end
	table.clear(self.Cleanups)
	for _, conn in ipairs(self.Connections) do
		conn:Disconnect()
	end
	table.clear(self.Connections)
	self.Gui:Destroy()
end

---------------------------------------------------------------- config
local function cleanName(name)
	return (string.gsub(tostring(name or ""), "[^%w_%- ]", ""))
end

function Library:SaveConfig(name)
	local clean = cleanName(name)
	if clean == "" then
		return false
	end
	local data = {}
	for k, v in pairs(self.Flags) do
		if string.sub(k, 1, 1) ~= "_" then
			data[k] = v
		end
	end
	local json = HttpService:JSONEncode(data)
	if hasFS() then
		return (pcall(function()
			if not isfolder(FOLDER) then
				makefolder(FOLDER)
			end
			writefile(FOLDER .. "/" .. clean .. ".json", json)
		end))
	end
	self.Memory[clean] = json
	return true
end

function Library:LoadConfig(name)
	local clean = cleanName(name)
	local json
	if hasFS() then
		pcall(function()
			json = readfile(FOLDER .. "/" .. clean .. ".json")
		end)
	else
		json = self.Memory[clean]
	end
	if not json then
		return false
	end
	local ok, data = pcall(function()
		return HttpService:JSONDecode(json)
	end)
	if not ok or type(data) ~= "table" then
		return false
	end
	for k, v in pairs(data) do
		local el = self.Elements[k]
		if el and string.sub(k, 1, 1) ~= "_" then
			el:Set(v)
		end
	end
	return true
end

function Library:DeleteConfig(name)
	local clean = cleanName(name)
	if hasFS() and typeof(delfile) == "function" then
		return (pcall(function()
			delfile(FOLDER .. "/" .. clean .. ".json")
		end))
	end
	self.Memory[clean] = nil
	return true
end

function Library:ListConfigs()
	local names = {}
	if hasFS() and typeof(listfiles) == "function" and isfolder(FOLDER) then
		pcall(function()
			for _, path in ipairs(listfiles(FOLDER)) do
				local n = string.match(path, "([^/\\]+)%.json$")
				if n then
					table.insert(names, n)
				end
			end
		end)
	else
		for n in pairs(self.Memory) do
			table.insert(names, n)
		end
	end
	table.sort(names)
	return names
end

---------------------------------------------------------------- tabs
local function slug(s)
	return (string.lower(tostring(s)):gsub("[^%w]", ""))
end

function Library:TabNames()
	local names = {}
	for _, tab in ipairs(self.Tabs) do
		table.insert(names, tab.Name)
	end
	return names
end

function Library:_findTab(name)
	local key = slug(name)
	for _, tab in ipairs(self.Tabs) do
		if slug(tab.Name) == key then
			return tab
		end
	end
	return nil
end

function Library:_drawTabIcon(tab, spec)
	local parts, holder = makeIcon(tab.Button, spec)
	if not parts then
		return false
	end
	if tab.IconHolder then
		tab.IconHolder:Destroy()
	end
	tab.IconHolder, tab.IconParts = holder, parts
	local tint = (self.CurrentTab == tab) and C.Text or C.Dim
	for _, part in ipairs(parts) do
		part[1][part[2]] = tint
	end
	return true
end

function Library:SetTabIcon(target, spec)
	local tab = target
	if type(target) == "string" then
		tab = self:_findTab(target)
	end
	if not tab then
		return false
	end
	local mode, value = parseIconSpec(spec)
	if mode == "kind" and not ICON_SET[value] then
		return false
	end
	local flag = "icon_" .. tab.Name
	local el = self.Elements[flag]
	if not el then
		el = makeValue(self, flag, spec, function(v)
			self:_drawTabIcon(tab, v)
		end)
	end
	el:Set(spec, true)
	return true
end

local function makeKeybind(win, flag, parent, position, default)
	local chip = new("TextButton", {
		AnchorPoint = Vector2.new(1, 0.5), Position = position, Size = UDim2.fromOffset(62, 24),
		BackgroundColor3 = "@Bg", Text = "", Face = "Mono", TextSize = 11, TextColor3 = "@Dim",
		AutoButtonColor = false, Parent = parent,
	}, { corner(6), stroke("@Line", 1) })
	local st = chip:FindFirstChildOfClass("UIStroke")
	local listening = false
	local obj
	obj = makeValue(win, flag, "", function(v)
		chip.Text = listening and "..." or shortKey(v)
		tween(st, 0.15, { Color = listening and C.Accent or C.Line })
		tween(chip, 0.15, { TextColor3 = listening and C.Text or C.Dim })
	end)
	obj._pressed = {}
	function obj:OnPressed(fn)
		table.insert(self._pressed, fn)
		return self
	end
	local baseSet = obj.Set
	function obj:Set(v, silent)
		if typeof(v) == "EnumItem" then
			v = v.Name
		end
		if v ~= nil and v ~= "" and not resolveKey(v) then
			return
		end
		baseSet(self, v or "", silent)
	end
	obj:Set(default, true)

	win:_connect(chip.MouseButton1Click, function()
		listening = true
		obj:Set(obj.Value, true)
	end)
	win:_connect(UserInputService.InputBegan, function(input, gp)
		if listening then
			local picked
			if input.UserInputType == Enum.UserInputType.Keyboard then
				picked = (input.KeyCode == Enum.KeyCode.Escape) and "" or input.KeyCode.Name
			elseif input.UserInputType == Enum.UserInputType.MouseButton2 or input.UserInputType == Enum.UserInputType.MouseButton3 then
				picked = input.UserInputType.Name
			end
			if picked ~= nil then
				listening = false
				obj:Set(picked)
			end
			return
		end
		if gp then
			return
		end
		local k = resolveKey(obj.Value)
		if k and (input.KeyCode == k or input.UserInputType == k) then
			for _, fn in ipairs(obj._pressed) do
				task.spawn(fn)
			end
		end
	end)
	return obj
end

		Position = UDim2.new(0, 6, 0, 26), Size = UDim2.new(1, -12, 0, 32), BackgroundTransparency = 1, Text = "",
		AutoButtonColor = false, Parent = row,
	})

	local obj = makeValue(win, flag, default, function(v)
		local a = (max == min) and 0 or (v - min) / (max - min)
		tween(fill, 0.1, { Size = UDim2.fromScale(a, 1) })
		tween(knob, 0.1, { Position = UDim2.fromScale(a, 0.5) })

		end
	end)

		if dragging and isMove(input) then
			update(input.Position.X)
		end
	end)
	win:_connect(UserInputService.InputEnded, function(input)
		if dragging and isPress(input) then
			dragging = false
			self.Page.ScrollingEnabled = true
			tween(knob, 0.2, { Size = UDim2.fromOffset(14, 14) })
			showBubble(false)
		end
	end)
	return obj

function Tab:AddDropdown(opts)
	local win = self.Window
	local flag = opts.Flag or opts.Name
	local options = opts.Options or {}
	local multi = opts.Multi == true
	local searchable = opts.Search == true or #options > 7
	local HEAD, ITEM = 40, 28
	local SEARCH = searchable and 34 or 0
	local open = false

	local holder = newRow(self, HEAD, opts.Name)
	holder.ClipsDescendants = true
	win:_tip(holder, opts.Tooltip)

	local head = new("TextButton", {
		Size = UDim2.new(1, 0, 0, HEAD), BackgroundTransparency = 1, Text = "", AutoButtonColor = false, Parent = holder,
	})
	newLabel(head, opts.Name)
	local selected = text({
		AnchorPoint = Vector2.new(1, 0), Position = UDim2.new(1, -36, 0, 0), Size = UDim2.new(0.4, 0, 1, 0), Text = "",
		Face = "Med", TextSize = 12, TextColor3 = "@Dim", TextXAlignment = Enum.TextXAlignment.Right,
		TextTruncate = Enum.TextTruncate.AtEnd, Parent = head,
	})
	local arrow = new("Frame", {
		AnchorPoint = Vector2.new(1, 0.5), Position = UDim2.new(1, -12, 0.5, 0), Size = UDim2.fromOffset(14, 14),
		BackgroundTransparency = 1, Parent = head,
	}, {
		new("Frame", {
			AnchorPoint = Vector2.new(0.5, 0.5), Position = UDim2.new(0.5, -3, 0.5, 0), Size = UDim2.fromOffset(9, 2),
			Rotation = 45, BackgroundColor3 = "@Accent", BorderSizePixel = 0,
		}),
		new("Frame", {
			AnchorPoint = Vector2.new(0.5, 0.5), Position = UDim2.new(0.5, 3, 0.5, 0), Size = UDim2.fromOffset(9, 2),
			Rotation = -45, BackgroundColor3 = "@Accent", BorderSizePixel = 0,
		}),
	})

	local searchBox
	if searchable then
		searchBox = new("TextBox", {
			Position = UDim2.fromOffset(8, HEAD + 2), Size = UDim2.new(1, -16, 0, 26), BackgroundColor3 = "@Bg",
			Text = "", PlaceholderText = "buscar...", PlaceholderColor3 = "@Dim", Face = "Med", TextSize = 12,
			TextColor3 = "@Text", ClearTextOnFocus = false, TextXAlignment = Enum.TextXAlignment.Left, Parent = holder,
		}, { corner(6), stroke("@Line", 1), padding(8, 0, 8, 0) })
	end

	local listFrame = new("ScrollingFrame", {
		Position = UDim2.fromOffset(0, HEAD + SEARCH), Size = UDim2.new(1, 0, 1, -(HEAD + SEARCH)),
		BackgroundTransparency = 1, BorderSizePixel = 0, ScrollBarThickness = 2, ScrollBarImageColor3 = "@Accent",
		CanvasSize = UDim2.new(), AutomaticCanvasSize = Enum.AutomaticSize.Y, Parent = holder,
	}, { listLayout(0), padding(4, 0, 4, 4) })

	local obj
	local buttons, dots, optionConns = {}, {}, {}

	local function isSel(name)
		if multi then
			return type(obj.Value) == "table" and table.find(obj.Value, name) ~= nil
		end
		return obj.Value == name
	end
	local function fullHeight()
		return HEAD + SEARCH + 8 + math.min(#options, 5) * ITEM
	end
	local function setOpen(state)
		open = state
		if state then
			if win.OpenDropdown and win.OpenDropdown ~= obj then
				win.OpenDropdown:Close()
			end
			win.OpenDropdown = obj
		elseif win.OpenDropdown == obj then
			win.OpenDropdown = nil
		end
		tween(holder, 0.28, { Size = UDim2.new(1, 0, 0, state and fullHeight() or HEAD) }, Enum.EasingStyle.Quint)
		tween(arrow, 0.25, { Rotation = state and 180 or 0 })
	end
	local function paint()
		if not obj then
			return
		end
		for name, b in pairs(buttons) do
			local sel = isSel(name)
			tween(b, 0.15, { TextColor3 = sel and C.Text or C.Dim })
			tween(dots[name], 0.15, { BackgroundTransparency = sel and 0 or 1, Size = sel and UDim2.fromOffset(7, 7) or UDim2.fromOffset(3, 3) })
		end
	end
	local function build()
		for _, conn in ipairs(optionConns) do
			conn:Disconnect()
		end
		table.clear(optionConns)
		for _, b in pairs(buttons) do
			b:Destroy()
		end
		table.clear(buttons)
		table.clear(dots)
		for i, name in ipairs(options) do
			local b = new("TextButton", {
				Size = UDim2.new(1, 0, 0, ITEM), LayoutOrder = i, BackgroundColor3 = "@Accent", BackgroundTransparency = 1,
				Text = tostring(name), Face = "Med", TextSize = 13, TextColor3 = "@Dim", TextXAlignment = Enum.TextXAlignment.Left,
				AutoButtonColor = false, Parent = listFrame,
			}, { corner(6), padding(10, 0, 0, 0) })
			dots[name] = new("Frame", {
				AnchorPoint = Vector2.new(1, 0.5), Position = UDim2.new(1, -10, 0.5, 0), Size = UDim2.fromOffset(3, 3),
				BackgroundColor3 = "@Accent", BackgroundTransparency = 1, BorderSizePixel = 0, Parent = b,
			}, { circleCorner() })
			table.insert(optionConns, b.MouseEnter:Connect(function()
				tween(b, 0.12, { BackgroundTransparency = 0.92 })
			end))
			table.insert(optionConns, b.MouseLeave:Connect(function()
				tween(b, 0.12, { BackgroundTransparency = 1 })
			end))
			table.insert(optionConns, b.MouseButton1Click:Connect(function()
				if multi then
					local list = table.clone(obj.Value or {})
					local idx = table.find(list, name)
					if idx then
						table.remove(list, idx)
					else
						table.insert(list, name)
					end
					obj:Set(list)
				else
					obj:Set(name)
					setOpen(false)
				end
			end))
			buttons[name] = b
		end
		paint()
		if open then
			tween(holder, 0.2, { Size = UDim2.new(1, 0, 0, fullHeight()) })
		end
	end
	win:_connect(holder.Destroying, function()
		for _, conn in ipairs(optionConns) do
			conn:Disconnect()
		end
	end)

	local default = opts.Default
	if multi then
		default = type(default) == "table" and default or {}
	elseif default == nil then
		default = options[1]
	end
	obj = makeValue(win, flag, default, function(v)
		if multi then
			local n = type(v) == "table" and #v or 0
			selected.Text = (n == 0 and "Nenhum") or (n == 1 and tostring(v[1])) or (n .. " selecionados")
		else
			selected.Text = tostring(v or "")
		end
		paint()
	end)
	function obj:Close()
		if open then
			setOpen(false)
		end
	end
	if multi then
		local baseSet = obj.Set
		function obj:Set(v, silent)
			baseSet(self, type(v) == "table" and v or {}, silent)
		end
	end
	function obj:Refresh(newOptions)
		options = newOptions or {}
		if not multi and self.Value ~= nil and not table.find(options, self.Value) then
			self:Set(nil, true)
		end
		build()
	end
	if opts.Callback then
		obj:OnChanged(opts.Callback)
	end
	build()
	obj:Set(default, true)
	table.insert(win.Restylers, paint)

	if searchBox then
		win:_connect(searchBox:GetPropertyChangedSignal("Text"), function()
			local q = string.lower(searchBox.Text)
			for name, b in pairs(buttons) do
				b.Visible = q == "" or string.find(string.lower(tostring(name)), q, 1, true) ~= nil
			end
		end)
	end
	win:_connect(head.MouseButton1Click, function()
		setOpen(not open)
	end)
	return obj
end

			c = value
		elseif type(value) == "string" then
			c, parsedAlpha = fromHex(value)
		end
		if not c then
			return
		end
		h, s, v = c:ToHSV()
		if useAlpha then
			alpha = math.clamp(a or parsedAlpha or alpha, 0, 1)
		end

function Tab:AddStats()

			TextColor3 = WHITE, TextXAlignment = Enum.TextXAlignment.Center, Parent = cell,
		})
		metalGradient(val, "text", 0)
		text({
			Position = UDim2.fromOffset(0, 38), Size = UDim2.new(1, 0, 0, 12), Text = d[1], Face = "Mono", TextSize = 9,
			TextColor3 = "@Dim", TextXAlignment = Enum.TextXAlignment.Center, Parent = cell,
		})
		cells[i] = val
		if i > 1 then
			new("Frame", {
				Position = UDim2.new((i - 1) / 4, 0, 0.5, 0), AnchorPoint = Vector2.new(0, 0.5), Size = UDim2.fromOffset(1, 34),
				BackgroundColor3 = "@Line", BorderSizePixel = 0, Parent = row,
			})
		end
	end
	local acc = 0
	local page = self.Page
	win:_connect(RunService.Heartbeat, function(dt)
		acc += dt
		if acc < 0.5 or not page.Visible or not win.Visible then
			return
		end
		acc = 0
		local t = round(os.clock() - win.StartTime)
		cells[1].Text = tostring(win.FPS)
		cells[2].Text = win.Ping .. "ms"
		cells[3].Text = string.format("%02d:%02d", math.floor(t / 60), t % 60)
		cells[4].Text = tostring(#Players:GetPlayers())
	end)

---------------------------------------------------------------- fundo por ID (ImageLabel dentro do Main)
function Library:SetBackground(source)
	if self.BgImage then
		self.BgImage:Destroy()
		self.BgImage = nil
	end
	if self.BgShade then
		self.BgShade:Destroy()
		self.BgShade = nil
	end
	local image = normalizeAssetId(source)
	if image then
		local img = new("ImageLabel", {
			Name = "Background", Position = UDim2.fromScale(0, 0), Size = UDim2.fromScale(1, 1),
			BackgroundTransparency = 1, BorderSizePixel = 0, Image = image,
			ScaleType = self.BgScale or Enum.ScaleType.Crop, ZIndex = 0, Active = false, Parent = self.Main,
		}, { corner(14) })
		self.BgImage = img
		self.BgShade = new("Frame", {
			Name = "BackgroundShade", Position = UDim2.fromScale(0, 0), Size = UDim2.fromScale(1, 1),
			BackgroundColor3 = "@Bg", BackgroundTransparency = 0.55, BorderSizePixel = 0, ZIndex = 0, Active = false,
			Parent = self.Main,
		}, { corner(14) })
		task.spawn(function()
			pcall(function()
				ContentProvider:PreloadAsync({ img }, function(_, status)
					if (status == Enum.AssetFetchStatus.Failure or status == Enum.AssetFetchStatus.TimedOut)
						and not self.Destroyed and self.BgImage == img then
						self:Notify("Fundo nao carregou: use o ID da imagem", "warn", 4)
					end
				end)
			end)
		end)
	end
	self:RefreshBackground()
	return image ~= nil
end

function Library:SetBackgroundDim(percent)
	self.BgDim = math.clamp(tonumber(percent) or 55, 0, 100)
	self:RefreshBackground()
end

function Library:SetBackgroundShade(percent)
	self.BgShadeP = math.clamp(tonumber(percent) or 45, 0, 100)
	self:RefreshBackground()
end

function Library:SetBackgroundScale(scaleType)
	self.BgScale = scaleType
	if self.BgImage then
		self.BgImage.ScaleType = scaleType
	end
end

function Library:RefreshBackground()
	local active = self.BgImage ~= nil
	local dim = (self.BgDim or 55) / 100
	local rowT = active and math.clamp(1 - dim - 0.1, 0, 1) or 0
	self.BgRowT = active and rowT or nil
	for _, row in ipairs(self.RowFrames) do
		if row.Parent then
			row.BackgroundTransparency = rowT
		end
	end
	if self.BgShade then
		self.BgShade.BackgroundTransparency = 1 - (self.BgShadeP or 45) / 100
	end
end

---------------------------------------------------------------- seletor de icones
function Library:ApplyIconPlan(plan)
	local targets = {}
	if plan.all then
		for _, tab in ipairs(self.Tabs) do
			table.insert(targets, tab)
		end
	elseif plan.tab then
		local tab = self:_findTab(plan.tab)
		if tab then
			table.insert(targets, tab)
		end
	end
	if #targets == 0 then
		return false, "Nao achei essa aba"
	end
	local done = 0
	for _, tab in ipairs(targets) do
		local spec = plan.reset and tab.DefaultIcon or plan.icon
		if spec and self:SetTabIcon(tab, spec) then
			done += 1
		end
	end
	if done == 0 then
		return false, "Icone invalido"
	end
	return true, "Icone atualizado em " .. done .. " aba(s)"
end

local function espFlag(path)
	return "esp_" .. string.gsub(path, "%.", "_")
end

local function espCoerce(e, value)
	if e.kind == "toggle" then
		if type(value) == "boolean" then
			return value
		end
		if value == 1 then
			return true
		elseif value == 0 then
			return false
		end
		return nil
	elseif e.kind == "dropdown" then
		return table.find(e.options, value) and value or nil
	elseif e.kind == "slider" then
		local n = tonumber(value)
		if not n then
			return nil
		end
		local step = e.step or 1
		return math.clamp(math.floor((n - e.min) / step + 0.5) * step + e.min, e.min, e.max)
	elseif e.kind == "color" then
		if typeof(value) == "Color3" then
			return value
		elseif type(value) == "string" then
			return (fromHex(value))
		end
	end
	return nil
end

local function segment(f, a, b, th)
	local d = b - a
	f.Size = UDim2.fromOffset(d.Magnitude, th)
	f.Position = UDim2.fromOffset((a.X + b.X) / 2, (a.Y + b.Y) / 2)
	f.Rotation = math.deg(math.atan2(d.Y, d.X))
end

local function tagLabel(parent, txt, faceName, size)
	return new("TextLabel", {
		AnchorPoint = Vector2.new(0.5, 0), Size = UDim2.fromOffset(0, 16), AutomaticSize = Enum.AutomaticSize.X,
		BackgroundColor3 = BLACK, BackgroundTransparency = 0.3, BorderSizePixel = 0, Text = txt, Face = faceName,
		TextSize = size, TextColor3 = WHITE, ZIndex = 5, Parent = parent,
	}, {
		corner(5), padding(7, 0, 7, 0),
		new("UIStroke", { Color = WHITE, Thickness = 1, Transparency = 0.85, ApplyStrokeMode = Enum.ApplyStrokeMode.Border }),
	})
end

	local tanH = math.tan(math.rad(FOVY) / 2)
	local function recalcCam()
		camCF = CFrame.lookAt(look.center + Vector3.new(0, 0, -look.height * 2.25), look.center)
	end
	recalcCam()
	local function project(v3)
		local rel = camCF:PointToObjectSpace(v3)
		local z = -rel.Z
		if z < 0.1 then
			return Vector2.new(SW / 2, SH / 2)
		end
		return Vector2.new((rel.X / (z * tanH * (SW / SH)) * 0.5 + 0.5) * SW, (0.5 - rel.Y / (z * tanH) * 0.5) * SH)
	end
	local function bounds()
		local top = project(look.center + Vector3.new(0, look.height / 2, 0))
		local bot = project(look.center - Vector3.new(0, look.height / 2, 0))
		local bh = (bot.Y - top.Y) + 10
		return SW / 2, (top.Y + bot.Y) / 2, bh * 0.5, bh
	end

	local function makeAvatarLayer(stage)
		local L = { angle = 0, model = nil, r15 = false, base = {}, chamsKey = "", dimState = nil }
		L.vp = new("ViewportFrame", {
			Size = UDim2.fromOffset(SW, SH), BackgroundTransparency = 1, Ambient = rgb(150, 150, 158), LightColor = WHITE,

		})

		L.vp.CurrentCamera = L.cam
		L.cam.CFrame = camCF
		L.silLines = {}
		for i = 1, #SK_SEGS do
			L.silLines[i] = new("Frame", {
				AnchorPoint = Vector2.new(0.5, 0.5), BackgroundColor3 = rgb(90, 92, 102), BorderSizePixel = 0, ZIndex = 2,
				Parent = stage,

			BorderSizePixel = 0, ZIndex = 2, Parent = stage,
		}, { circleCorner() })
		table.insert(layers, L)

	local function mount(L)
		if not template or L.model then
			return
		end
		local m = template:Clone()
		for _, d in ipairs(m:GetDescendants()) do
			if d:IsA("BasePart") then
				d.Anchored = true
				L.base[d] = { d.Color, d.Material, d.Transparency }
			end
		end
		m.Parent = L.vp
		m:PivotTo(CFrame.new(0, 0, 0))
		L.model = m
		L.r15 = m:FindFirstChild("UpperTorso") ~= nil
		L.cam.CFrame = camCF
	end

	local function joints(L)
		local pts = {}
		local m = L.model
		local rot = CFrame.Angles(0, L.angle, 0)
		for _, key in ipairs(SK_KEYS) do
			local p
			if m then
				if L.r15 then
					local part = m:FindFirstChild(SK_PARTS[key])
					if part then
						p = part.Position
					end
				else
					local spec = R6_MAP[key]
					local part = m:FindFirstChild(spec[1])
					if part then
						p = part.CFrame:PointToWorldSpace(Vector3.new(0, spec[2], 0))
					end
				end
			end
			if not p then
				p = look.center + rot:VectorToWorldSpace(MANNEQUIN[key])
			end
			pts[key] = project(p)
		end
		return pts
	end

	local function silhouette(L, pts, color, transparency)
		local show = L.model == nil
		for i, seg in ipairs(SK_SEGS) do
			local f = L.silLines[i]
			f.Visible = show
			if show then
				segment(f, pts[seg[1]], pts[seg[2]], SIL_TH[i] * (look.height / 5.8))
				f.BackgroundColor3, f.BackgroundTransparency = color, transparency
			end
		end
		L.silHead.Visible = show
		if show then
			local s = 30 * (look.height / 5.8)
			L.silHead.Size = UDim2.fromOffset(s, s)
			L.silHead.Position = UDim2.fromOffset(pts.head.X, pts.head.Y)
			L.silHead.BackgroundColor3, L.silHead.BackgroundTransparency = color, transparency
		end
	end

	local function advance(L, dt)
		if st["demo.spin"] then
			L.angle += dt * 0.9
			if L.model then
				L.model:PivotTo(CFrame.new(0, 0, 0) * CFrame.Angles(0, L.angle, 0))
			end
		end
	end

			if d:IsA("Script") or d:IsA("LocalScript") then
				d:Destroy()

					tBoxes[k].Position = UDim2.fromOffset(px, cyb)
					tBoxes[k]:FindFirstChildOfClass("UIStroke").Color = (k == 1) and BAD or col
					tTexts[k].Text = string.format("%dm", round(dist))
					if tLines[k].Visible then
						segment(tLines[k], origin, Vector2.new(px, gy), st["tracer.thickness"])
						tLines[k].BackgroundColor3 = tcol
						tLines[k].BackgroundTransparency = 0.25 + frac * 0.45
					end

					local right = rel > 0

					tArrows[k].BackgroundTransparency = 0.15 + math.abs(math.sin(t * 3 + k)) * 0.4
				end
			end
			decor(t)
		end
		return { stage = stage, render = render }
	end

				if el then
					el:Set(st[path], true)
				end
			end
		end
		return true
	end

function Tab:AddPlayerList()
	local win = self.Window
	local query = ""
	local rows, thumbs = {}, {}

	self:AddSection("Jogadores")
	local countLbl = self:AddLabel("0 jogadores")
	local searchEl = self:AddInput({ Name = "Buscar", Flag = "_tp_search", Placeholder = "nome do jogador" })
	self.Count += 1
	local container = new("Frame", {
		Size = UDim2.new(1, 0, 0, 0), AutomaticSize = Enum.AutomaticSize.Y, BackgroundTransparency = 1,
		LayoutOrder = self.Count, Parent = self.Page,
	}, { listLayout(6) })

	local function teleportTo(plr)
		local myChar, theirChar = LocalPlayer.Character, plr.Character
		local a = myChar and myChar:FindFirstChild("HumanoidRootPart")
		local b = theirChar and theirChar:FindFirstChild("HumanoidRootPart")
		if a and b then
			myChar:PivotTo(b.CFrame * CFrame.new(0, 0, 3))
			win:Notify("Teleportado para " .. plr.DisplayName, "success", 2)
		else
			win:Notify("Personagem indisponivel", "warn", 2)
		end
	end

	local function reorder()
		local list = {}
		for plr in pairs(rows) do
			table.insert(list, plr)
		end
		table.sort(list, function(a, b)
			return string.lower(a.DisplayName) < string.lower(b.DisplayName)
		end)
		local shown = 0
		local q = string.lower(query)
		for i, plr in ipairs(list) do
			local r = rows[plr]
			r.Frame.LayoutOrder = i
			local match = q == "" or string.find(string.lower(plr.DisplayName), q, 1, true) ~= nil
				or string.find(string.lower(plr.Name), q, 1, true) ~= nil
			r.Frame.Visible = match
			if match then
				shown += 1
			end
		end
		countLbl.Text = shown .. " de " .. #list .. " jogadores"
	end

	local function fetchThumb(plr, img)
		if thumbs[plr.UserId] then
			img.Image = thumbs[plr.UserId]
			return
		end
		task.spawn(function()
			local ok, url = pcall(function()
				return Players:GetUserThumbnailAsync(plr.UserId, Enum.ThumbnailType.HeadShot, Enum.ThumbnailSize.Size100x100)
			end)
			if ok and url then
				thumbs[plr.UserId] = url
				if img.Parent then
					img.Image = url
				end
			end
		end)
	end

	local function build(plr)
		if plr == LocalPlayer or rows[plr] then
			return
		end
		local r = new("Frame", {
			Size = UDim2.new(1, 0, 0, 56), BackgroundColor3 = WHITE, BorderSizePixel = 0, Parent = container,

			TextSize = 10, TextColor3 = "@Dim", TextTruncate = Enum.TextTruncate.AtEnd, Parent = r,
		})
		local dist = text({
			AnchorPoint = Vector2.new(1, 0.5), Position = UDim2.new(1, -112, 0.5, 0), Size = UDim2.fromOffset(48, 14),
			Text = "--", Face = "Mono", TextSize = 10, TextColor3 = "@Dim", TextXAlignment = Enum.TextXAlignment.Right, Parent = r,

---------------------------------------------------------------- aba Resolucao (Lighting)
local function buildResolution(tab, win)
	local orig = {
		Brightness = Lighting.Brightness, Exposure = Lighting.ExposureCompensation, Clock = Lighting.ClockTime,
	}
	local camera = Workspace.CurrentCamera
	local origFov = camera and camera.FieldOfView or 70

	local function fx(class)
		local e = Instance.new(class)
		e.Name = HttpService:GenerateGUID(false)
		e.Parent = Lighting
		return e
	end
	local cc = fx("ColorCorrectionEffect")
	local bloom = fx("BloomEffect")
	bloom.Intensity, bloom.Size, bloom.Threshold = 0, 24, 0.9
	local blur = fx("BlurEffect")
	blur.Size = 0
	local rays = fx("SunRaysEffect")
	rays.Intensity, rays.Spread = 0, 0.6

	local S = { bright = 0, contrast = 0, sat = 0, sharp = 0, blur = 0, bloom = 0, rays = 0, tint = WHITE }
	local function apply()
		cc.Brightness = S.bright / 100 * 0.5
		cc.Contrast = S.contrast / 100 + S.sharp / 100 * 0.22
		cc.Saturation = S.sat / 100
		cc.TintColor = S.tint
		bloom.Intensity = S.bloom / 100 * 1.5
		blur.Size = math.max(0, S.blur - S.sharp * 0.12)
		rays.Intensity = S.rays / 100
	end
	table.insert(win.Cleanups, function()
		cc:Destroy()
		bloom:Destroy()
		blur:Destroy()
		rays:Destroy()
		Lighting.Brightness, Lighting.ExposureCompensation, Lighting.ClockTime = orig.Brightness, orig.Exposure, orig.Clock
		if camera then
			camera.FieldOfView = origFov
		end
	end)

	local els = {}
	local function sl(name, flag, key, min, max, def, suffix, tip, step)
		local el = tab:AddSlider({ Name = name, Flag = flag, Min = min, Max = max, Default = def, Suffix = suffix, Tooltip = tip, Step = step })
		el:OnChanged(function(val)
			S[key] = val
			apply()
		end)
		els[key] = { el = el, def = def }
		return el
	end

	tab:AddSection("Presets")
	local PRESET_VALUES = {
		["Padrao"] = { bright = 0, contrast = 0, sat = 0, sharp = 0, blur = 0, bloom = 0, rays = 0 },
		["Vivido"] = { bright = 4, contrast = 14, sat = 38, sharp = 20, blur = 0, bloom = 12, rays = 0 },
		["Cinema"] = { bright = -6, contrast = 22, sat = -12, sharp = 10, blur = 0, bloom = 22, rays = 10 },
		["P&B"] = { bright = 0, contrast = 18, sat = -100, sharp = 15, blur = 0, bloom = 0, rays = 0 },
		["Noite"] = { bright = -16, contrast = 12, sat = -20, sharp = 0, blur = 0, bloom = 18, rays = 0 },
		["Nitido"] = { bright = 2, contrast = 10, sat = 14, sharp = 80, blur = 0, bloom = 0, rays = 0 },
	}
	tab:AddDropdown({
		Name = "Preset de imagem", Flag = "res_preset",
		Options = { "Padrao", "Vivido", "Cinema", "P&B", "Noite", "Nitido" }, Default = "Padrao",
	}):OnChanged(function(name)
		local p = PRESET_VALUES[name]
		if not p then
			return
		end
		for key, val in pairs(p) do
			if els[key] then
				els[key].el:Set(val)
			end
		end
	end)

	tab:AddSection("Cor e brilho")
	sl("Brilho", "res_bright", "bright", -100, 100, 0, "", "Brilho geral da imagem")
	sl("Contraste", "res_contrast", "contrast", -100, 100, 0, "", "Diferenca entre claros e escuros")
	sl("Saturacao", "res_sat", "sat", -100, 100, 0, "", "-100 = preto e branco")
	sl("Nitidez", "res_sharp", "sharp", 0, 100, 0, "%", "Simulada: contraste fino + menos desfoque")
	tab:AddColorPicker({ Name = "Cor de tonalidade", Flag = "res_tint", Default = WHITE }):OnChanged(function(c)
		S.tint = c
		apply()
	end)

	tab:AddSection("Efeitos")
	sl("Bloom", "res_bloom", "bloom", 0, 100, 0, "%", "Brilho suave em areas claras")
	sl("Desfoque", "res_blur", "blur", 0, 30, 0, "", "Desfoque de tela")
	sl("Raios de sol", "res_rays", "rays", 0, 100, 0, "%", "Raios de luz do sol")

	tab:AddSection("Mundo")
	tab:AddSlider({ Name = "Exposicao", Flag = "res_exposure", Min = -3, Max = 3, Step = 0.1, Default = 0 })
		:OnChanged(function(v)
			Lighting.ExposureCompensation = v
		end)
	tab:AddSlider({ Name = "Luz do mundo", Flag = "res_world", Min = 0, Max = 6, Step = 0.1, Default = math.clamp(orig.Brightness, 0, 6) })
		:OnChanged(function(v)
			Lighting.Brightness = v
		end)
	tab:AddSlider({ Name = "Horario", Flag = "res_clock", Min = 0, Max = 24, Step = 0.1, Default = math.clamp(orig.Clock, 0, 24) })
		:OnChanged(function(v)
			Lighting.ClockTime = v
		end)
	tab:AddSlider({ Name = "Campo de visao (FOV)", Flag = "res_fov", Min = 30, Max = 120, Default = math.clamp(round(origFov), 30, 120) })
		:OnChanged(function(v)
			local cam = Workspace.CurrentCamera
			if cam then
				cam.FieldOfView = v
			end
		end)

	tab:AddSection("Interface")
	tab:AddSlider({ Name = "Escala da UI", Flag = "res_ui_scale", Min = 55, Max = 110, Default = round(win.BaseScale * 100), Suffix = "%" })
		:OnChanged(function(v)
			win:SetScale(v / 100)
		end)

	tab:AddButton({ Name = "Resetar imagem", Style = "Accent" }):OnClick(function()
		for _, info in pairs(els) do
			info.el:Set(info.def)
		end
		win.Elements.res_tint:Set(WHITE)
		win.Elements.res_exposure:Set(0)
		win.Elements.res_world:Set(orig.Brightness)
		win.Elements.res_clock:Set(orig.Clock)
		win.Elements.res_fov:Set(origFov)
		win:Notify("Imagem restaurada", "success", 1.8)
	end)
end

--
-- v8.2: acessorios (chapeu, cabelo, asas, costas, etc.) agora sao copiados SEMPRE, em separado,
--       porque o metodo nativo no cliente so troca roupa/cores.
--
-- Como a copia funciona (3 camadas, para nao dar erro):
--   1) Humanoid:ApplyDescriptionReset / ApplyDescription com a HumanoidDescription do alvo
--   2) se nao refletir no personagem: transplanta acessorios, roupas, cores e rosto
--      a partir de um snapshot (personagem do jogador, ou modelo montado pela descricao)
--   3) se a conta nao devolver descricao: Players:GetCharacterAppearanceAsync
--
-- USO (arquivo separado):
--   local SkinChanger = loadstring(readfile("SkinChanger.lua"))()
--   local tab = Window:AddTab("Skin Changer", "user")
--   SkinChanger(Window, tab, { new=new, corner=corner, circleCorner=circleCorner, stroke=stroke,
--       text=text, tween=tween, metalGradient=metalGradient, C=C, WHITE=WHITE })
--

local Players = game:GetService("Players")
local InsertService = game:GetService("InsertService")
local LocalPlayer = Players.LocalPlayer

local SCALE_NAMES = {
	"BodyHeightScale", "BodyWidthScale", "BodyDepthScale", "HeadScale", "BodyTypeScale", "BodyProportionScale",
}
local WEARABLE = { Shirt = true, Pants = true, ShirtGraphic = true, BodyColors = true, CharacterMesh = true }
local SKIP_PART = { HumanoidRootPart = true }

---------------------------------------------------------------- helpers (sem UI)
local function trim(s)
	return (string.gsub(tostring(s or ""), "^%s*(.-)%s*$", "%1"))
end

local function try(fn, attempts)
	local total = attempts or 2
	local lastErr
	for i = 1, total do
		local ok, result = pcall(fn)
		if ok and result ~= nil then
			return result
		end
		lastErr = result
		if i < total then
			task.wait(0.4)
		end
	end
	return nil, lastErr
end

local function destroy(inst)
	if inst then
		pcall(function()
			inst:Destroy()
		end)
	end
end

local function safeClone(inst)
	local ok, c = pcall(function()
		return inst:Clone()
	end)
	if ok and c then
		return c
	end
	local made
	pcall(function()
		local was = inst.Archivable
		inst.Archivable = true
		made = inst:Clone()
		inst.Archivable = was
	end)
	return made
end

local function getHumanoid(timeout)
	local deadline = os.clock() + (timeout or 0)
	while true do
		local ch = LocalPlayer.Character
		local h = ch and ch:FindFirstChildOfClass("Humanoid")
		if h and h.Parent then
			return h, ch
		end
		if os.clock() >= deadline then
			return nil, nil
		end
		task.wait(0.1)
	end
end

-- snapshot = copia solta (sem parent) da aparencia de um personagem/modelo
local function snapshot(model, hum)
	local snap = { accessories = {}, extras = {}, colors = {}, scales = {}, face = nil }
	if not model then
		return snap
	end
	for _, child in ipairs(model:GetChildren()) do
		if child:IsA("Accessory") then
			local c = safeClone(child)
			if c then
				for _, d in ipairs(c:GetDescendants()) do
					if d.Name == "AccessoryWeld" then
						d:Destroy()
					end
				end
				table.insert(snap.accessories, c)
			end
		elseif WEARABLE[child.ClassName] then
			local c = safeClone(child)
			if c then
				table.insert(snap.extras, c)
			end
		elseif child:IsA("BasePart") and not SKIP_PART[child.Name] then
			snap.colors[child.Name] = child.Color
		end
	end
	local head = model:FindFirstChild("Head")
	local decal = head and head:FindFirstChildOfClass("Decal")
	if decal then
		snap.face = decal.Texture
	end
	local humanoid = hum or model:FindFirstChildOfClass("Humanoid")
	if humanoid then
		for _, name in ipairs(SCALE_NAMES) do
			local v = humanoid:FindFirstChild(name)
			if v and v:IsA("NumberValue") then
				snap.scales[name] = v.Value
			end
		end
	end
	return snap
end

local function destroySnap(snap)
	if not snap then
		return
	end
	for _, inst in ipairs(snap.accessories) do
		destroy(inst)
	end
	for _, inst in ipairs(snap.extras) do
		destroy(inst)
	end
	table.clear(snap.accessories)
	table.clear(snap.extras)
end

-- confere se a descricao realmente chegou no personagem
local function looksApplied(desc)
	local h, ch = getHumanoid()
	if not h or not ch then
		return false
	end
	local ok, result = pcall(function()
		local expected = #desc:GetAccessories(true)
		local have = 0
		for _, c in ipairs(ch:GetChildren()) do
			if c:IsA("Accessory") then
				have += 1
			end
		end
		if expected > 0 and have == 0 then
			return false
		end
		if desc.Shirt ~= 0 and not ch:FindFirstChildOfClass("Shirt") then
			return false
		end
		return true
	end)
	return ok and result == true
end

-- camada 1: API nativa
local function applyNative(desc)
	local h = getHumanoid(3)
	if not h then
		return false, "sem personagem"
	end
	local lastErr
	for _, method in ipairs({ "ApplyDescriptionReset", "ApplyDescription" }) do
		local ok, err = pcall(function()
			h[method](h, desc)
		end)
		if ok then
			task.wait(0.15)
			if looksApplied(desc) then
				return true
			end
			lastErr = "nao refletiu no personagem"
		else
			lastErr = tostring(err)
		end
	end
	return false, lastErr
end

-- camada 2: transplante a partir do snapshot
local function applySnapshot(snap)
	local h, ch = getHumanoid(3)
	if not h or not ch then
		return false, "sem personagem"
	end
	local ok, err = pcall(function()
		for _, child in ipairs(ch:GetChildren()) do
			if child:IsA("Accessory") or WEARABLE[child.ClassName] then
				child:Destroy()
			end
		end
		for _, inst in ipairs(snap.extras) do
			local c = inst:Clone()
			if c then
				c.Parent = ch
			end
		end
		for _, acc in ipairs(snap.accessories) do
			local c = acc:Clone()
			if c then
				local attached = pcall(function()
					h:AddAccessory(c)
				end)
				if not attached or not c.Parent then
					c.Parent = ch
				end
			end
		end
		for _, part in ipairs(ch:GetChildren()) do
			local col = part:IsA("BasePart") and snap.colors[part.Name]
			if col then
				part.Color = col
			end
		end
		if snap.face then
			local head = ch:FindFirstChild("Head")
			local decal = head and head:FindFirstChildOfClass("Decal")
			if decal then
				decal.Texture = snap.face
			end
		end
		for name, val in pairs(snap.scales) do
			local v = h:FindFirstChild(name)
			if v and v:IsA("NumberValue") then
				v.Value = val
			end
		end
	end)
	if not ok then
		return false, tostring(err)
	end
	return true
end

-- troca TODOS os acessorios do personagem pela lista (clones soltos)
local function replaceAccessories(list)
	local h, ch = getHumanoid(3)
	if not h or not ch then
		return false
	end
	local ok = pcall(function()
		for _, child in ipairs(ch:GetChildren()) do
			if child:IsA("Accessory") then
				child:Destroy()
			end
		end
		for _, acc in ipairs(list) do
			local c = acc:Clone()
			if c then
				local attached = pcall(function()
					h:AddAccessory(c)
				end)
				if not attached or not c.Parent then
					c.Parent = ch
				end
			end
		end
	end)
	return ok
end

-- baixa os acessorios da descricao direto do catalogo (hat, cabelo, rosto, costas, cintura...)
local function loadAccessories(desc)
	local list = {}
	local ok, accs = pcall(function()
		return desc:GetAccessories(true)
	end)
	if not ok or type(accs) ~= "table" then
		return list
	end
	for _, a in ipairs(accs) do
		local id = a.AssetId
		if id and id ~= 0 then
			local model = try(function()
				return InsertService:LoadAsset(id)
			end, 2)
			if model then
				local acc = model:FindFirstChildWhichIsA("Accessory", true)
				local c = acc and safeClone(acc)
				if c then
					for _, d in ipairs(c:GetDescendants()) do
						if d.Name == "AccessoryWeld" then
							d:Destroy()
						end
					end
					table.insert(list, c)
				end
				destroy(model)
			end
		end
	end
	return list
end

-- monta o snapshot sob demanda (camadas 2 e 3)
local function ensureSnap(look)
	if look.snap then
		return look.snap
	end
	local h = getHumanoid()
	local rig = h and h.RigType or Enum.HumanoidRigType.R15
	if look.desc then
		local model = try(function()
			return Players:CreateHumanoidModelFromDescription(look.desc, rig)
		end, 2)
		if model then
			look.snap = snapshot(model)
			destroy(model)
			return look.snap
		end
	end
	if look.userId then
		local model = try(function()
			return Players:GetCharacterAppearanceAsync(look.userId)
		end, 2)
		if model then
			look.snap = snapshot(model)
			destroy(model)
			return look.snap
		end
	end
	return nil
end

-- look = { desc = HumanoidDescription?, snap = snapshot?, userId = number? }
local function applyLook(look)
	local h = getHumanoid(4)
	if not h then
		return false, "personagem indisponivel"
	end
	if h.Health <= 0 then
		return false, "voce esta morto"
	end
	local why
	local nativeOk = false
	if look.desc then
		local ok, err = applyNative(look.desc)
		nativeOk = ok
		if not ok then
			why = err
		end
	end

	-- acessorios: sempre copiados em separado (o nativo no cliente so troca roupa)
	local snap = ensureSnap(look)
	local accs = snap and snap.accessories or {}
	local expected = 0
	if look.desc then
		local okc, list = pcall(function()
			return look.desc:GetAccessories(true)
		end)
		if okc and type(list) == "table" then
			expected = #list
		end
	end
	if #accs == 0 and expected > 0 then
		accs = loadAccessories(look.desc)
		if #accs > 0 then
			look.snap = look.snap or { accessories = {}, extras = {}, colors = {}, scales = {}, face = nil }
			look.snap.accessories = accs
			snap = look.snap
		end
	end
	local accOk = false
	if #accs > 0 or (snap ~= nil and expected == 0) then
		accOk = replaceAccessories(accs)
	end

	if nativeOk then
		return true, "nativo"
	end
	if snap then
		local ok, err = applySnapshot(snap)
		if ok then
			return true, "manual"
		end
		why = err
	end
	if accOk then
		return true, "acessorios"
	end
	return false, why or "avatar indisponivel"
end

---------------------------------------------------------------- modulo
local function SkinChanger(Window, tab, ctx)
	local new, corner, circleCorner, stroke = ctx.new, ctx.corner, ctx.circleCorner, ctx.stroke
	local text, tween, metalGradient = ctx.text, ctx.tween, ctx.metalGradient
	local WHITE = ctx.WHITE

	local state = {
		source = "Jogo", -- "Jogo" = roupa atual no servidor | "Roblox" = avatar salvo na conta
		keep = true,
		look = nil, -- ultima skin aplicada (reaplicada ao renascer)
		originalDesc = nil,
		originalSnap = nil,
		changed = false,
		busy = false,
	}

	local function notify(msg, kind)
		if not Window.Destroyed then
			Window:Notify(msg, kind, 2.6)
		end
	end

	local function discard(look)
		if look then
			destroy(look.desc)
			destroySnap(look.snap)
		end
	end

	local function setLook(look)
		local prev = state.look
		state.look = look
		if prev and prev ~= look then
			discard(prev)
		end
	end

	-- guarda a skin original para poder restaurar
	task.spawn(function()
		local h, ch = getHumanoid(20)
		if not h or Window.Destroyed then
			return
		end
		local ok, d = pcall(function()
			return h:GetAppliedDescription()
		end)
		if ok and d then
			state.originalDesc = d
		end
		state.originalSnap = snapshot(ch, h)
	end)

	local function buildLook(info)
		local look = { name = info.display, userId = info.userId }
		if state.source == "Jogo" and info.player then
			local ch = info.player.Character
			local h = ch and ch:FindFirstChildOfClass("Humanoid")
			if h then
				local ok, d = pcall(function()
					return h:GetAppliedDescription()
				end)
				if ok and d then
					look.desc = d
				end
				look.snap = snapshot(ch, h)
			end
		end
		if not look.desc then
			look.desc = try(function()
				return Players:GetHumanoidDescriptionFromUserId(info.userId)
			end, 2)
		end
		return look
	end

	---------------------------------------------------------------- UI superior
	tab:AddSection("Skin Changer")
	local status = tab:AddLabel("Skin atual: a sua")

	tab:AddSelector({
		Name = "Fonte da skin", Flag = "skin_source", Options = { "Jogo", "Roblox" }, Default = "Jogo",
		Tooltip = "Jogo = roupa atual no servidor | Roblox = avatar salvo na conta",
	}):OnChanged(function(v)
		state.source = v
	end)

	tab:AddToggle({
		Name = "Manter ao renascer", Flag = "skin_keep", Default = true,
		Description = "Reaplica a skin depois que voce morrer",
	}):OnChanged(function(v)
		state.keep = v
	end)

	tab:AddButton({ Name = "Restaurar minha skin", Style = "Accent" }):OnClick(function()
		if state.busy then
			notify("Aguarde, aplicando outra skin", "warn")
			return
		end
		if not state.originalDesc and not state.originalSnap then
			notify("Skin original ainda nao carregou", "warn")
			return
		end
		state.busy = true
		task.spawn(function()
			local ok, how = applyLook({
				desc = state.originalDesc, snap = state.originalSnap, userId = LocalPlayer.UserId,
			})
			state.busy = false
			if Window.Destroyed then
				return
			end
			if ok then
				setLook(nil)
				state.changed = false
				status.Text = "Skin atual: a sua"
				notify("Skin restaurada", "success")
			else
				notify("Nao consegui: " .. string.sub(tostring(how), 1, 50), "error")
			end
		end)
	end)

	---------------------------------------------------------------- card de jogador
	local thumbs = {}

	local function fetchThumb(userId, img)
		if thumbs[userId] then
			img.Image = thumbs[userId]
			return
		end
		task.spawn(function()
			local url
			for _, kind in ipairs({ Enum.ThumbnailType.AvatarBust, Enum.ThumbnailType.HeadShot }) do
				local ok, content = pcall(function()
					return (Players:GetUserThumbnailAsync(userId, kind, Enum.ThumbnailSize.Size100x100))
				end)
				if ok and type(content) == "string" and content ~= "" then
					url = content
					break
				end
			end
			if url then
				thumbs[userId] = url
				if img.Parent then
					img.Image = url
				end
			end
		end)
	end

	local function copyFrom(info, btn)
		if state.busy then
			notify("Aguarde, aplicando outra skin", "warn")
			return
		end
		state.busy = true
		local old = btn.Text
		btn.Text = "..."
		local look = buildLook(info)
		local ok, how = applyLook(look)
		state.busy = false
		if btn.Parent then
			btn.Text = old
		end
		if Window.Destroyed then
			discard(look)
			return
		end
		if ok then
			setLook(look)
			state.changed = true
			status.Text = "Skin atual: " .. info.display
			notify("Avatar de " .. info.display .. " copiado", "success")
		else
			discard(look)
			notify("Nao consegui: " .. string.sub(tostring(how), 1, 50), "error")
		end
	end

	local function makeCard(parent, info)
		local r = new("Frame", {
			Size = UDim2.new(1, 0, 0, 60), BackgroundColor3 = WHITE, BorderSizePixel = 0, Parent = parent,
		}, { corner(9), stroke("@Line", 1) })
		metalGradient(r, "bg", 0)

		local img = new("ImageLabel", {
			Position = UDim2.fromOffset(9, 10), Size = UDim2.fromOffset(40, 40), BackgroundColor3 = "@Track",
			BorderSizePixel = 0, Image = "", Parent = r,
		}, { circleCorner() })
		local ims = new("UIStroke", {
			Color = WHITE, Thickness = 1.8, ApplyStrokeMode = Enum.ApplyStrokeMode.Border, Parent = img,
		})
		Window:_wave(metalGradient(ims, "metal", 45), 1.2, info.userId % 7, 1)

		local nm = text({
			Position = UDim2.fromOffset(60, 11), Size = UDim2.new(1, -184, 0, 18), Text = info.display, Face = "Bold",
			TextSize = 13, TextColor3 = WHITE, TextTruncate = Enum.TextTruncate.AtEnd, Parent = r,
		})
		metalGradient(nm, "text", 0)
		text({
			Position = UDim2.fromOffset(60, 31), Size = UDim2.new(1, -184, 0, 14), Text = "@" .. info.name, Face = "Mono",
			TextSize = 10, TextColor3 = "@Dim", TextTruncate = Enum.TextTruncate.AtEnd, Parent = r,
		})

		local btn = new("TextButton", {
			AnchorPoint = Vector2.new(1, 0.5), Position = UDim2.new(1, -10, 0.5, 0), Size = UDim2.fromOffset(108, 30),
			BackgroundColor3 = WHITE, Text = "COPY AVATAR", Face = "Black", TextSize = 9, TextColor3 = "@Bg",
			AutoButtonColor = false, Parent = r,
		}, { corner(7) })
		Window:_wave(metalGradient(btn, "bright", 0), 1.4, info.userId % 5, 1)

		local st = r:FindFirstChildOfClass("UIStroke")
		Window:_connect(r.MouseEnter, function()
			tween(st, 0.15, { Color = ctx.C.Dim })
		end)
		Window:_connect(r.MouseLeave, function()
			tween(st, 0.15, { Color = ctx.C.Line })
		end)
		Window:_connect(btn.MouseButton1Click, function()
			tween(btn, 0.08, { Size = UDim2.fromOffset(100, 28) })
			task.delay(0.1, function()
				if btn.Parent then
					tween(btn, 0.18, { Size = UDim2.fromOffset(108, 30) }, Enum.EasingStyle.Back)
				end
			end)
			task.spawn(copyFrom, info, btn)
		end)

		fetchThumb(info.userId, img)
		return r
	end

	local function newContainer()
		tab.Count += 1
		return new("Frame", {
			Size = UDim2.new(1, 0, 0, 0), AutomaticSize = Enum.AutomaticSize.Y, BackgroundTransparency = 1,
			LayoutOrder = tab.Count, Parent = tab.Page,
		}, { new("UIListLayout", { SortOrder = Enum.SortOrder.LayoutOrder, Padding = UDim.new(0, 6) }) })
	end

	---------------------------------------------------------------- jogadores do servidor
	tab:AddSection("Jogadores do servidor")
	local countLbl = tab:AddLabel("0 jogadores")
	local filterEl = tab:AddInput({ Name = "Filtrar", Flag = "_skin_filter", Placeholder = "nome do jogador" })
	local listBox = newContainer()
	local rows, query = {}, ""

	local function reorder()
		local list = {}
		for plr in pairs(rows) do
			table.insert(list, plr)
		end
		table.sort(list, function(a, b)
			return string.lower(a.DisplayName) < string.lower(b.DisplayName)
		end)
		local q, shown = string.lower(query), 0
		for i, plr in ipairs(list) do
			local f = rows[plr]
			f.LayoutOrder = i
			local match = q == "" or string.find(string.lower(plr.DisplayName), q, 1, true) ~= nil
				or string.find(string.lower(plr.Name), q, 1, true) ~= nil
			f.Visible = match
			if match then
				shown += 1
			end
		end
		countLbl.Text = shown .. " de " .. #list .. " jogadores"
	end

	local function addPlayer(plr)
		if plr == LocalPlayer or rows[plr] then
			return
		end
		rows[plr] = makeCard(listBox, {
			userId = plr.UserId, name = plr.Name, display = plr.DisplayName, player = plr,
		})
		reorder()
	end

	for _, plr in ipairs(Players:GetPlayers()) do
		addPlayer(plr)
	end
	Window:_connect(Players.PlayerAdded, addPlayer)
	Window:_connect(Players.PlayerRemoving, function(plr)
		local f = rows[plr]
		if f then
			f:Destroy()
			rows[plr] = nil
			reorder()
		end
	end)
	filterEl:OnChanged(function(v)
		query = tostring(v or "")
		reorder()
	end)

	---------------------------------------------------------------- pesquisar qualquer conta do Roblox
	tab:AddSection("Pesquisar no Roblox")
	tab:AddInput({ Name = "Usuario ou ID", Flag = "_skin_query", Placeholder = "ex: Roblox ou 1" })
	local resultBox = newContainer()
	local searching = false

	local function clearResults()
		for _, c in ipairs(resultBox:GetChildren()) do
			if c:IsA("GuiObject") then
				c:Destroy()
			end
		end
	end

	local function search()
		if searching then
			return
		end
		local q = trim(Window.Flags._skin_query)
		q = string.gsub(q, "^@", "")
		if q == "" then
			notify("Digite um usuario ou ID", "warn")
			return
		end
		searching = true
		task.spawn(function()
			local userId
			if string.match(q, "^%d+$") then
				userId = tonumber(q)
			else
				userId = try(function()
					return Players:GetUserIdFromNameAsync(q)
				end, 2)
			end
			local username
			if userId then
				username = try(function()
					return Players:GetNameFromUserIdAsync(userId)
				end, 2)
			end
			searching = false
			if Window.Destroyed then
				return
			end
			clearResults()
			if not userId or not username then
				notify("Usuario nao encontrado", "error")
				return
			end
			makeCard(resultBox, { userId = userId, name = username, display = username, player = nil })
			notify("Encontrado: " .. username, "success")
		end)
	end

	tab:AddButton({ Name = "Pesquisar skin", Style = "Accent" }):OnClick(search)

	---------------------------------------------------------------- reaplicar ao renascer + limpeza
	Window:_connect(LocalPlayer.CharacterAdded, function(ch)
		if not state.keep or not state.look then
			return
		end
		local loaded = false
		local conn = LocalPlayer.CharacterAppearanceLoaded:Connect(function()
			loaded = true
		end)
		local t0 = os.clock()
		while not loaded and os.clock() - t0 < 4 and not Window.Destroyed do
			task.wait(0.1)
		end
		conn:Disconnect()
		task.wait(0.3)
		if Window.Destroyed or state.busy or not state.keep or not state.look then
			return
		end
		if LocalPlayer.Character ~= ch then
			return
		end
		state.busy = true
		applyLook(state.look)
		state.busy = false
	end)

	table.insert(Window.Cleanups, function()
		local look = state.look
		state.look = nil
		local orig = { desc = state.originalDesc, snap = state.originalSnap, userId = LocalPlayer.UserId }
		task.spawn(function()
			if state.changed and (orig.desc or orig.snap) then
				pcall(applyLook, orig)
			end
			destroy(orig.desc)
			destroySnap(orig.snap)
			discard(look)
		end)
	end)

	return state
end

	local function polyPoints(n, rr, rotDeg, startDeg)
		local pts = {}
		for i = 0, n - 1 do

			pts[i + 1] = Vector2.new(CX + math.cos(a) * rr, CY + math.sin(a) * rr)
		end
		return pts
	end

	local function edges(pts, closed, th, tr, t, dashed, f0, f1)
		local n = #pts
		local count = closed and n or (n - 1)
		for i = 1, count do
			local a = pts[i]
			local b = closed and pts[i % n + 1] or pts[i + 1]
			local col = colorAt(f0 + (f1 - f0) * (i - 1) / count, t)
			if dashed then
				local len = (b - a).Magnitude
				local k = math.floor(len / 9)
				if k <= 1 then
					if i % 2 == 1 then
						line(a, b, th, col, tr)
					end
				else
					for j = 0, k - 1 do
						if j % 2 == 0 then
							line(a + (b - a) * (j / k), a + (b - a) * ((j + 1) / k), th, col, tr)
						end
					end
				end
			else
				line(a, b, th, col, tr)
			end
		end
	end

	local function arc(startDeg, spanDeg, rr, th, tr, t, f0, f1)
		local n = math.max(2, math.ceil(spanDeg / 10) + 1)
		local pts = {}
		for i = 0, n - 1 do
			local a = math.rad(startDeg + spanDeg * i / (n - 1))
			pts[i + 1] = Vector2.new(CX + math.cos(a) * rr, CY + math.sin(a) * rr)
		end
		edges(pts, false, th, tr, t, false, f0, f1)
	end

	local function drawShape(t, R)
		local style, th = st.style, st.thick
		local tr = 1 - st.opacity / 100
		if st.pulse then
			tr = math.clamp(tr + (0.5 + 0.5 * math.sin(t * 4)) * 0.3, 0, 0.9)
		end
		local rot = angle
		local fillKind, fillR = "circle", R
		local dashed = st.dash

		local function poly(n, startDeg, rr)
			local pts = polyPoints(n, rr or R, rot, startDeg)
			edges(pts, true, th, tr, t, dashed, 0, 1)
			return pts
		end

		if style == "Circulo" then
			if dashed then
				poly(48, -90)
			else
				setRing(rings[1], R, th, tr, t)
			end
		elseif style == "Aneis Duplos" then
			setRing(rings[1], R, th, tr, t)
			setRing(rings[2], R * 0.82, math.max(1, th - 1), math.min(tr + 0.15, 0.95), t)
		elseif style == "Losango" then
			poly(4, -90)
			fillKind, fillR = "diamond", R * 0.7071
		elseif style == "Quadrado" then
			poly(4, -45)
			fillKind, fillR = "square", R * 0.7071
		elseif style == "Triangulo" then
			poly(3, -90)
			fillR = R * 0.5
		elseif style == "Hexagono" then
			poly(6, -90)
			fillR = R * 0.866
		elseif style == "Octogono" then
			poly(8, -90)
			fillR = R * 0.924
		elseif style == "Poligono" then
			local n = st.sides

			fillR = R * math.cos(math.pi / n)
		elseif style == "Estrela" then
			local pts = {}
			for i = 0, 9 do
				local rr = (i % 2 == 0) and R or R * 0.45
				local a = math.rad(-90 + rot + i * 36)
				pts[i + 1] = Vector2.new(CX + math.cos(a) * rr, CY + math.sin(a) * rr)
			end
			edges(pts, true, th, tr, t, dashed, 0, 1)

			sweep.BackgroundColor3, trail.BackgroundColor3 = st.c1, st.c1
		elseif style == "Segmentado" then

				local dir = Vector2.new(math.cos(a), math.sin(a))
				line(Vector2.new(CX, CY) + dir * (R - 5), Vector2.new(CX, CY) + dir * (R + 9), th, colorAt(i / 4, t), tr)
			end
		elseif style == "Arcos" then
			for i = 0, 3 do
				arc(-90 + rot + i * 90 - 26, 52, R, th, tr, t, i / 4, (i + 1) / 4)
			end
		elseif style == "Pulsante" then
			setRing(rings[1], R, th, tr, t)
		end

		if st.ticks > 0 then
			local n = st.ticks
			for i = 0, n - 1 do
				local a = math.rad(rot + i * 360 / n - 90)
				local dir = Vector2.new(math.cos(a), math.sin(a))
				local len = (i % 3 == 0) and 10 or 5
				line(Vector2.new(CX, CY) + dir * (R + 3), Vector2.new(CX, CY) + dir * (R + 3 + len), 1, colorAt(i / n, t), math.min(tr + 0.1, 0.95))
			end
		end

		if st.fill > 0 then
			fill.Visible = true
			fill.Position = UDim2.fromOffset(CX, CY)
			fill.Size = UDim2.fromOffset(fillR * 2, fillR * 2)
			fill.BackgroundColor3 = st.c1
			fill.BackgroundTransparency = 1 - st.fill / 100
			if fillKind == "circle" then
				fillCorner.CornerRadius = UDim.new(0.5, 0)
				fill.Rotation = 0
			elseif fillKind == "square" then
				fillCorner.CornerRadius = UDim.new(0, 0)
				fill.Rotation = rot
			else
				fillCorner.CornerRadius = UDim.new(0, 0)
				fill.Rotation = 45 + rot
			end
		end

		if st.glow then
			local th2 = 8 + (st.pulse and math.sin(t * 4) * 2 or 0)
			plainRing(aura, R + 4, th2, 0.86 + (st.pulse and math.sin(t * 4) * 0.06 or 0), colorAt(0.25, t))
		end

		if st.ripple or style == "Pulsante" then
			for k = 1, 3 do
				local p = (t * 0.7 + k / 3) % 1
				plainRing(ripples[k], R * (0.25 + 0.9 * p), 2, 0.35 + 0.65 * p, colorAt(p, t))
			end
		end
	end

	local function drawTargets(t, R)
		local n = st.targets
		local sp = st.tspeed / 100
		local inside, nearest, nearestD, nearestIdx = 0, nil, math.huge, 0
		local positions = {}
		for i = 1, 12 do
			local d = tgt[i]
			if i <= n then
				local x = CX + math.sin(t * 0.55 * sp + i * 1.7) * (SW * 0.36) * (0.55 + 0.45 * math.cos(i * 0.9))
				local y = CY + math.sin(t * 0.8 * sp + i * 2.3) * (SH * 0.36)
				local dist = math.sqrt((x - CX) ^ 2 + (y - CY) ^ 2)
				local isIn = dist <= R
				positions[i] = Vector2.new(x, y)
				d.Visible = true
				d.Position = UDim2.fromOffset(x, y)
				d.BackgroundColor3 = (st.hilite and isIn) and st.inside or st.outside
				if isIn then
					inside += 1
					if dist < nearestD then
						nearestD, nearest, nearestIdx = dist, positions[i], i
					end
				end
				d.Size = UDim2.fromOffset(8, 8)
			else
				d.Visible = false
			end
		end
		if nearestIdx > 0 then
			tgt[nearestIdx].Size = UDim2.fromOffset(12, 12)
		end
		if st.snap and nearest then
			snapLine.Visible = true
			segment(snapLine, Vector2.new(CX, CY), nearest, 1)
			snapLine.BackgroundColor3 = st.inside
			snapLine.BackgroundTransparency = 0.3
		end
		return inside
	end

	local function drawCross()
		local ctype, sz, cc = st.cross, st.csize, st.ccolor
		if ctype == "Ponto" then
			chDot.Visible = true
			chDot.Position = UDim2.fromOffset(CX, CY)
			chDot.BackgroundColor3 = cc
			chDot.Size = UDim2.fromOffset(4, 4)
		elseif ctype == "Cruz" or ctype == "Cruz com vao" then
			local gap = (ctype == "Cruz") and 0 or 3
			local off = gap + sz / 2
			local defs = {
				{ CX, CY - off, 2, sz }, { CX + off, CY, sz, 2 }, { CX, CY + off, 2, sz }, { CX - off, CY, sz, 2 },
			}
			for i, d in ipairs(defs) do
				arms[i].Visible = true
				arms[i].Position = UDim2.fromOffset(d[1], d[2])
				arms[i].Size = UDim2.fromOffset(d[3], d[4])
				arms[i].BackgroundColor3 = cc
			end
		elseif ctype == "Circulo" then
			plainRing(chRing, sz, 1.5, 0, cc)
			chDot.Visible = true
			chDot.Position = UDim2.fromOffset(CX, CY)
			chDot.BackgroundColor3 = cc
			chDot.Size = UDim2.fromOffset(2, 2)
		end
	end

	local navTitle -- definido abaixo
	local lastNav = ""

	local function render(t, dt)
		local sw = clip.AbsoluteSize.X / math.max(Window.UIScale.Scale, 0.01)
		if sw < 60 then
			return
		end
		SW = sw
		CX, CY = SW / 2, SH / 2
		angle += dt * st.rot

		local R = 12 + (st.fov - 10) / 490 * (SH / 2 - 22 - 12)
		if st.breathe then
			R *= 1 + 0.04 * math.sin(t * 2.2)
		end

		for _, r in ipairs(rings) do
			r.f.Visible = false
		end
		aura.f.Visible = false
		for _, r in ipairs(ripples) do
			r.f.Visible = false
		end
		chRing.f.Visible = false
		fill.Visible = false
		sweep.Visible, trail.Visible = false, false
		snapLine.Visible = false
		for i = 1, 4 do
			arms[i].Visible = false
		end
		chDot.Visible = false
		used = 0

		drawShape(t, R)
		for i = used + 1, lastUsed do
			lines[i].Visible = false
		end
		lastUsed = used

		local inside = drawTargets(t, R)
		drawCross()

		if st.label then
			tagLbl.Visible = true
			tagLbl.Text = "FOV " .. tostring(st.fov)
			tagLbl.Position = UDim2.fromOffset(CX, math.max(34, CY - R - 20))
		else
			tagLbl.Visible = false
		end

		infoLbl.Text = string.format(
			"FOV %d  |  %s  |  %s  |  %d/%d DENTRO",
			st.fov, string.upper(st.style), string.upper(st.colormode), inside, st.targets
		)
		scan.Position = UDim2.fromOffset(0, ((t * 55) % (SH + 20)) - 10)
		liveDot.BackgroundTransparency = (math.sin(t * 4) > 0) and 0 or 0.6

		local idx = table.find(STYLES, st.style) or 1
		local nav = string.format("%02d/%02d  %s", idx, #STYLES, string.upper(st.style))
		if nav ~= lastNav and navTitle then
			lastNav = nav
			navTitle.Text = nav
		end
	end

	---------------------------------------------------------------- barra de navegacao
	local NAVY = SH + 8 + 24

	local function chevron(parent, dir)
		for _, s in ipairs({ -1, 1 }) do
			new("Frame", {
				AnchorPoint = Vector2.new(0.5, 0.5), Position = UDim2.new(0.5, dir * 2, 0.5, s * 3),
				Size = UDim2.fromOffset(9, 2), Rotation = dir * s * 45 * -1, BackgroundColor3 = "@Accent",
				BorderSizePixel = 0, Parent = parent,
			})
		end
	end

	local function hover(btn)
		local s = btn:FindFirstChildOfClass("UIStroke")
		Window:_connect(btn.MouseEnter, function()
			tween(s, 0.15, { Color = C.Accent })
		end)
		Window:_connect(btn.MouseLeave, function()
			tween(s, 0.15, { Color = C.Line })
		end)
	end

		}, { corner(7), stroke("@Line", 1) })

	end

	---------------------------------------------------------------- controles
	local els = {}
	for _, e in ipairs(SCHEMA) do
		if e.section then
			tab:AddSection(e.section)
		else
			local flag = "fov_" .. e.key
			local el
			if e.kind == "toggle" then
				el = tab:AddToggle({ Name = e.label, Flag = flag, Default = e.default })
			elseif e.kind == "dropdown" then
				el = tab:AddDropdown({ Name = e.label, Flag = flag, Options = e.options, Default = e.default })
			elseif e.kind == "slider" then
				el = tab:AddSlider({
					Name = e.label, Flag = flag, Min = e.min, Max = e.max, Default = e.default, Step = e.step, Suffix = e.suffix,
				})
			elseif e.kind == "color" then
				el = tab:AddColorPicker({ Name = e.label, Flag = flag, Default = e.default })
			end
			els[e.key] = el
			local key = e.key
			el:OnChanged(function(val)
				st[key] = val
			end)
		end
	end

	local function setKey(key, val)
		st[key] = val
		local el = els[key]
		if el then
			el:Set(val, true)
		end
	end

	local function applyPreset(name)
		local p = PRESETS[name]
		if not p then
			return false
		end
		for key, val in pairs(DEFAULTS) do
			setKey(key, val)
		end
		for key, val in pairs(p) do
			setKey(key, val)
		end
		return true
	end

	local function randomize()
		local pick = function(list)
			return list[math.random(#list)]
		end
		setKey("style", pick(STYLES))
		setKey("colormode", pick(COLOR_MODES))
		setKey("c1", Color3.fromHSV(math.random(), 0.5, 1))
		setKey("c2", Color3.fromHSV(math.random(), 0.7, 1))
		setKey("thick", math.random(1, 4))
		setKey("rot", pick({ -90, -45, 0, 0, 45, 90 }))
		setKey("glow", math.random() < 0.6)
		setKey("pulse", math.random() < 0.3)
		setKey("dash", math.random() < 0.2)
		setKey("ripple", math.random() < 0.3)
		setKey("ticks", pick({ 0, 0, 12, 24, 36 }))
		setKey("fill", pick({ 0, 0, 10, 20 }))
		setKey("segs", math.random(3, 12) * 2)
		setKey("sides", math.random(3, 12))
	end

	tab:AddSection("Presets")
	tab:AddDropdown({ Name = "Preset de FOV", Flag = "_fov_preset", Options = PRESET_ORDER, Default = "Limpo" })
		:OnChanged(function(name)
			applyPreset(name)
		end)

	local function stepStyle(delta)
		local idx = table.find(STYLES, st.style) or 1
		idx = (idx - 1 + delta) % #STYLES + 1
		setKey("style", STYLES[idx])
	end
	Window:_connect(prevBtn.MouseButton1Click, function()
		stepStyle(-1)
	end)
	Window:_connect(nextBtn.MouseButton1Click, function()
		stepStyle(1)
	end)
	Window:_connect(randBtn.MouseButton1Click, randomize)
	Window:_connect(resetBtn.MouseButton1Click, function()
		applyPreset("Limpo")
	end)

	---------------------------------------------------------------- loop
	local acc = 0
	Window:_connect(RunService.RenderStepped, function(dt)
		if not Window.Visible or Window.Minimized or not tab.Page.Visible then
			return
		end
		acc += dt
		if acc < 1 / 45 then
			return
		end
		local step = acc
		acc = 0
		render(os.clock(), step)
	end)

	return { State = st, Apply = applyPreset, Randomize = randomize, Styles = STYLES }
end
